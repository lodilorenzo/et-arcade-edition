# Reuse upstream's authoritative file lists; do not execute its Python-2 scripts.
function(et_sources output script variable directory)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/engine/src/${script}")
    file(READ "${CMAKE_SOURCE_DIR}/engine/src/${script}" text)
    string(REGEX MATCH "${variable} = \"\"\"([^\"]*)\"\"\"" match "${text}")
    if(match)
        set(body "${CMAKE_MATCH_1}")
    else()
        string(REGEX MATCH "${variable} = \\[([^]]*)\\]" match "${text}")
        if(NOT match)
            message(FATAL_ERROR "Missing upstream source list: ${variable}")
        endif()
        string(REGEX REPLACE "#[^\n]*" "" body "${CMAKE_MATCH_1}")
        string(REPLACE "," " " body "${body}")
    endif()
    separate_arguments(files UNIX_COMMAND "${body}")
    list(TRANSFORM files PREPEND "${CMAKE_SOURCE_DIR}/engine/src/${directory}/")
    set(${output} ${files} PARENT_SCOPE)
endfunction()

foreach(module game cgame ui)
    if(module STREQUAL game)
        set(variable qagame_string)
    else()
        set(variable "${module}_string")
    endif()
    et_sources(sources "SConscript.${module}" "${variable}" "${module}")
    if(module STREQUAL game)
        et_sources(botai SConscript.game botai_string botai)
        list(APPEND sources ${botai}
            "${CMAKE_SOURCE_DIR}/native/omnibot.cpp" "${CMAKE_SOURCE_DIR}/native/local_profiles.c"
            "${CMAKE_SOURCE_DIR}/vendor/omnibot/Common/BotLoadLibrary.cpp")
    endif()
    if(module STREQUAL cgame)
        list(APPEND sources "${CMAKE_SOURCE_DIR}/native/gamepad_aim.c")
    endif()
    add_library(${module} SHARED ${sources} "engine/src/${module}/${module}.def")
    if(module STREQUAL game)
        target_compile_definitions(${module} PRIVATE GAMEDLL)
        target_compile_features(${module} PRIVATE cxx_std_17)
        target_include_directories(${module} PRIVATE native engine/src/game vendor/omnibot/Common vendor/omnibot/ET)
        set(name qagame)
    elseif(module STREQUAL cgame)
        target_compile_definitions(${module} PRIVATE CGAMEDLL)
        set(name cgame)
    else()
        target_compile_definitions(${module} PRIVATE UIDLL)
        set(name ui)
    endif()
    set_target_properties(${module} PROPERTIES OUTPUT_NAME "${name}_mp_x86")
    target_link_libraries(${module} PRIVATE winmm)
    if(ET_RUST_RANDOM)
        target_compile_definitions(${module} PRIVATE ET_RUST_RANDOM)
    endif()
    if(ET_RUST_SHARED)
        target_compile_definitions(${module} PRIVATE ET_RUST_SHARED)
    endif()
    if(ET_RUST_RANDOM OR ET_RUST_SHARED)
        target_link_libraries(${module} PRIVATE rust_core)
    endif()
endforeach()

# Map-list script checks use the game's own action/event names (native/map_list.c).
file(READ "${CMAKE_SOURCE_DIR}/engine/src/game/g_script.c" et_script_source)
string(REGEX REPLACE "//[^\n]*" "" et_script_source "${et_script_source}")
set(et_script_names "")
foreach(table Actions Events)
    string(REGEX MATCH "gScript${table}\\[\\] =[^;]*" et_table "${et_script_source}")
    string(REGEX MATCHALL "\"[^\"]+\"" et_names "${et_table}")
    list(LENGTH et_names et_count)
    if(et_count LESS 10)
        message(FATAL_ERROR "Could not read gScript${table} from g_script.c")
    endif()
    string(REPLACE ";" ", " et_names "${et_names}")
    string(APPEND et_script_names "static const char *const etScript${table}[] = { ${et_names}, NULL };\n")
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/generated/et_script_names.h" "/* Generated from engine/src/game/g_script.c. */\n${et_script_names}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/engine/src/game/g_script.c")
